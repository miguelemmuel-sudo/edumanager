const sessionStr = localStorage.getItem('edu_session'); if (!sessionStr) return; const session = JSON.parse(sessionStr); const plan = session.plan || 'starter'; const maxEleves = plan === 'starter' ? 300 : (plan === 'standard' ? 1000 : (plan === 'premium' ? 1500 : Infinity)); if (maxEleves !== Infinity) { try { // Query Supabase for total number of students const { count, error } = await window.supabase .from('eleves') .select('*', { count: 'exact', head: true }) .eq('etablissement_id', session.etablissement_id); if (!error && count !== null && count >= maxEleves) { alert(`Vous avez atteint la limite de ${maxEleves} élèves pour le plan ${plan.toUpperCase()}. Veuillez mettre à niveau votre abonnement pour ajouter plus d'élèves.`); window.location.href = 'paiements.html'; return; } } catch(e) { console.error('Erreur vérification limites:', e); } } // Si tout est ok, ouvrir le modal const addModal = new bootstrap.Modal(document.getElementById('addEleveModal')); addModal.show(); }
